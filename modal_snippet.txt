      {/* MODAL 3: MANAGE THEME PROMPTS */}
      {showManageModal && selectedThemeForManage && (
        <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="w-full max-w-2xl bg-slate-900 border border-slate-800 rounded-3xl p-6 shadow-2xl space-y-5 animate-scale-up max-h-[85vh] flex flex-col">
            <div className="flex items-center justify-between pb-3 border-b border-slate-800 shrink-0">
              <div className="flex items-center gap-2">
                <FileText className="w-5 h-5 text-indigo-400" />
                <h3 className="text-base font-black text-white">
                  Manage Prompts: {selectedThemeForManage.name}
                </h3>
              </div>
              <button
                type="button"
                onClick={() => {
                  setShowManageModal(false);
                  setSelectedThemeForManage(null);
                }}
                className="p-1 rounded-lg text-slate-400 hover:text-white hover:bg-slate-800 transition"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <div className="overflow-y-auto pr-1 space-y-4 flex-1">
              <div className="p-4 bg-slate-950/60 border border-slate-800/80 rounded-2xl">
                <div className="text-xs font-bold text-slate-400 uppercase mb-2">Theme Details</div>
                <div className="text-sm font-semibold text-white">Slug: <span className="font-mono text-indigo-300">{selectedThemeForManage.slug}</span></div>
                <div className="text-sm font-semibold text-white">Category: <span className="text-slate-300">{selectedThemeForManage.category}</span></div>
              </div>

              <div className="space-y-3">
                <div className="flex items-center justify-between">
                  <div className="text-xs uppercase font-bold text-slate-400 tracking-wider">Active Scene Prompts</div>
                  <button
                    type="button"
                    onClick={() => {
                      setSelectedThemeForPrompt(selectedThemeForManage);
                      setShowAddPromptModal(true);
                    }}
                    className="px-3 py-1 bg-orange-500 hover:bg-orange-600 text-white rounded-lg text-xs font-bold transition flex items-center gap-1"
                  >
                    + Add Another Prompt
                  </button>
                </div>

                <div className="p-4 bg-slate-950 border border-slate-800 rounded-2xl text-xs text-slate-400 space-y-2">
                  <p>All scene prompts assigned to <strong className="text-white">{selectedThemeForManage.name}</strong> will be displayed and manageable here.</p>
                </div>
              </div>
            </div>

            <div className="pt-3 border-t border-slate-800 flex justify-end shrink-0">
              <button
                type="button"
                onClick={() => {
                  setShowManageModal(false);
                  setSelectedThemeForManage(null);
                }}
                className="px-5 py-2 bg-slate-800 hover:bg-slate-700 text-white text-xs font-bold rounded-xl transition"
              >
                Close
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
